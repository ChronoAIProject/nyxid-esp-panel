# For more information about build system see
# https://docs.espressif.com/projects/esp-idf/en/latest/api-guides/build-system.html
# The following five lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.5)
if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.rev1_3.defaults;sdkconfig.board-16mb.defaults")
endif()
set(EXTRA_COMPONENT_DIRS "$ENV{IDF_PATH}/examples/ethernet/basic/components/ethernet_init")
set(NYX_PANEL_EXPERIMENTAL_WAKE_MODEL "" CACHE FILEPATH "Private experimental Hey Nyx INT8 artifact")
set(NYX_PANEL_EXPERIMENTAL_WAKE_CALIBRATION "" CACHE FILEPATH "Validation calibration identifying the experimental artifact")
set(NYX_PANEL_EXPERIMENTAL_WAKE_SELFTEST "" CACHE FILEPATH "Optional private exact-PCM/output diagnostic for the experimental model")
set(NYX_PANEL_EXPERIMENTAL_WAKE_CANDIDATE_PROFILE "" CACHE FILEPATH "Explicit frozen teacher60 research profile; empty pins original v30")
set(NYX_PANEL_RESEARCH_HIESP_INPUT "" CACHE STRING "Research Hi ESP control input: raw or processed; empty keeps the normal build")
option(NYX_PANEL_RESEARCH_WAKE_TIMING "Measure the development wake detector's wall time" OFF)
set(NYX_PANEL_RESEARCH_WAKE_CLOCK_PCM "" CACHE FILEPATH "Optional offline PCM for the raw Hi ESP sample-clock probe")
if(NYX_PANEL_RESEARCH_WAKE_TIMING AND (NOT NYX_PANEL_NODE_DEV_MODE OR
        (NOT NYX_PANEL_RESEARCH_HIESP_INPUT AND NOT NYX_PANEL_EXPERIMENTAL_WAKE_MODEL)))
    message(FATAL_ERROR "Wake timing requires an isolated development wake arm")
endif()
if(NYX_PANEL_RESEARCH_WAKE_CLOCK_PCM AND (NOT NYX_PANEL_RESEARCH_WAKE_TIMING OR
        NOT NYX_PANEL_RESEARCH_HIESP_INPUT STREQUAL "raw"))
    message(FATAL_ERROR "Wake clock PCM requires the isolated raw Hi ESP timing arm")
endif()
set_property(CACHE NYX_PANEL_RESEARCH_HIESP_INPUT PROPERTY STRINGS "" "raw" "processed")
if(NOT "${NYX_PANEL_RESEARCH_HIESP_INPUT}" STREQUAL "")
    if(NOT NYX_PANEL_RESEARCH_HIESP_INPUT MATCHES "^(raw|processed)$")
        message(FATAL_ERROR "The research Hi ESP input must be raw or processed")
    endif()
    if(NOT NYX_PANEL_NODE_DEV_MODE)
        message(FATAL_ERROR "The Hi ESP research baseline requires isolated node development mode")
    endif()
    if(NYX_PANEL_EXPERIMENTAL_WAKE_MODEL)
        message(FATAL_ERROR "The Hi ESP baseline and experimental Hey Nyx model are separate research arms")
    endif()
endif()
if(NYX_PANEL_RESEARCH_HIESP_INPUT OR NYX_PANEL_EXPERIMENTAL_WAKE_MODEL)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/research/panel_wake_runtime/dependency_pins")
endif()
if(NYX_PANEL_EXPERIMENTAL_WAKE_MODEL)
    if(NOT NYX_PANEL_EXPERIMENTAL_WAKE_CALIBRATION)
        message(FATAL_ERROR "An experimental wake model requires its validation calibration")
    endif()
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/research/panel_wake_runtime")
elseif(NYX_PANEL_EXPERIMENTAL_WAKE_CALIBRATION OR NYX_PANEL_EXPERIMENTAL_WAKE_SELFTEST OR
        NYX_PANEL_EXPERIMENTAL_WAKE_CANDIDATE_PROFILE)
    message(FATAL_ERROR "A wake calibration requires its model")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PANEL_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
set(PROJECT_VER "${PANEL_VERSION}")
set(EXCLUDE_COMPONENTS brookesia_app_squareline_demo)
set(ENV{NYX_PANEL_PROJECT_DIR} "${CMAKE_CURRENT_SOURCE_DIR}")
project(nyxid-esp-panel)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/apply_managed_component_patches.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/panel_lcd_compat.cmake")
# Component Manager resolves environment variables in local lock paths. Keep
# those paths portable while retaining every registry version/hash in the lock.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock" _panel_lock)
    string(REPLACE "${CMAKE_CURRENT_SOURCE_DIR}/" "$NYX_PANEL_PROJECT_DIR/" _panel_lock "${_panel_lock}")
    file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock" "${_panel_lock}")
endif()
