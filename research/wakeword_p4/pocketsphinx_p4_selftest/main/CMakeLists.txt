set(work "${CMAKE_CURRENT_LIST_DIR}/../../../../.local/wakeword/c161-lexical-p4")
set(ps "${work}/pocketsphinx-5.0.4")
file(STRINGS "${ps}/src/CMakeLists.txt" upstream REGEX "^[a-z0-9_/]+[.]c$")
set(sources "main.c" "port.c")
foreach(file IN LISTS upstream)
    if(NOT file STREQUAL "util/mmio.c" AND NOT file STREQUAL "util/profile.c")
        list(APPEND sources "${ps}/src/${file}")
    endif()
endforeach()
set(gen "${CMAKE_CURRENT_BINARY_DIR}/psconfig")
set(SIZEOF_LONG 4)
set(SIZEOF_LONG_LONG 8)
set(HAVE_LONG_LONG 1)
set(HAVE_SNPRINTF 1)
set(HAVE_SYS_STAT_H 1)
set(HAVE_SYS_TYPES_H 1)
set(HAVE_UNISTD_H 1)
set(HAVE_INTTYPES_H 1)
set(HAVE_STDINT_H 1)
set(PACKAGE_NAME pocketsphinx)
set(PACKAGE_VERSION 5.0.4)
set(PACKAGE_STRING "PocketSphinx 5.0.4")
configure_file("${ps}/config.h.in" "${gen}/config.h")
configure_file("${ps}/sphinx_config.h.in" "${gen}/pocketsphinx/sphinx_config.h")
# P4 newlib uses long for int32_t. Upstream APIs also pass int pointers directly.
file(READ "${ps}/include/pocketsphinx/prim_type.h" prim)
string(REGEX REPLACE "typedef int32_t[ 	]+int32;" "_Static_assert(sizeof(int) == 4, \"PocketSphinx int32 ABI\");
typedef int int32;" prim "${prim}")
string(REGEX REPLACE "typedef uint32_t[ 	]+uint32;" "typedef unsigned int uint32;" prim "${prim}")
file(WRITE "${gen}/pocketsphinx/prim_type.h" "${prim}")
# ESP-IDF provides elapsed time but has no getrusage system call.
file(READ "${ps}/src/util/profile.c" profile)
string(REPLACE "(! defined(_HPUX_SOURCE))" "(! defined(ESP_PLATFORM) && ! defined(_HPUX_SOURCE))" profile "${profile}")
file(WRITE "${gen}/profile.c" "${profile}")
list(APPEND sources "${gen}/profile.c")
if(NYX_LEXICAL_Q8)
    list(REMOVE_ITEM sources "${ps}/src/ms_gauden.c")
    file(READ "${ps}/src/ms_gauden.c" gauden)
    string(REPLACE "if ((fp = fopen(file_name, \"rb\")) == NULL)" "void *owner = NULL;
    if ((fp = packed_gaussian_open(file_name, &owner)) == NULL)" gauden "${gauden}")
    string(REPLACE "fclose(fp);" "packed_gaussian_close(fp, owner);" gauden "${gauden}")
    file(WRITE "${gen}/ms_gauden.c" "#include \"packed_gaussian.h\"
${gauden}")
    list(APPEND sources "${gen}/ms_gauden.c" "packed_gaussian.c")
endif()
idf_component_register(SRCS ${sources} PRIV_INCLUDE_DIRS "." "${gen}" "${ps}/include" "${ps}/src" PRIV_REQUIRES fatfs esp_timer heap esp_partition spiffs)
target_compile_definitions(${COMPONENT_LIB} PRIVATE HAVE_CONFIG_H)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-error)
set(volume "${work}")
if(NYX_LEXICAL_COMPACT)
    set(volume "${work}/compact")
endif()
if(NYX_LEXICAL_Q8)
    fatfs_create_rawflash_image(cases "${work}/q8/cases" FLASH_IN_PROJECT)
    spiffs_create_partition_image(storage "${work}/q8/volume" FLASH_IN_PROJECT)
    set_source_files_properties(main.c PROPERTIES COMPILE_DEFINITIONS NYX_LEXICAL_Q8)
else()
    fatfs_create_rawflash_image(storage "${volume}/files" FLASH_IN_PROJECT)
endif()

if(NYX_LEXICAL_FLASH_MAP)
    idf_build_get_property(python PYTHON)
    add_custom_target(nyx_map_offsets
        COMMAND "${python}" "${CMAKE_CURRENT_LIST_DIR}/../map_offsets.py"
                "${CMAKE_BINARY_DIR}/storage.bin" "${volume}" "${gen}/map_offsets.inc"
        BYPRODUCTS "${gen}/map_offsets.inc"
        VERBATIM)
    add_dependencies(nyx_map_offsets fatfs_storage_bin)
    add_dependencies(${COMPONENT_LIB} nyx_map_offsets)
    set_source_files_properties(main.c port.c PROPERTIES COMPILE_DEFINITIONS NYX_LEXICAL_FLASH_MAP)
endif()
