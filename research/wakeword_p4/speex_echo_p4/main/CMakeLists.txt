set(src "${CMAKE_CURRENT_LIST_DIR}/../../../../.local/wakeword/c161-first-command-v5/lexical-echo/speexdsp-1.2.1")
set(dsp "${src}/libspeexdsp")
if(NYX_SPEEX_FLOAT)
    set(fft_src "${dsp}/smallft.c")
    set(arithmetic FLOATING_POINT USE_SMALLFT)
else()
    set(fft_src "${dsp}/kiss_fft.c" "${dsp}/kiss_fftr.c")
    set(arithmetic FIXED_POINT USE_KISS_FFT)
endif()
idf_component_register(SRCS "main.c" "${dsp}/mdf.c" "${dsp}/fftwrap.c" ${fft_src}
                       PRIV_INCLUDE_DIRS "." "${src}/include" PRIV_REQUIRES esp_timer spi_flash heap)
target_compile_definitions(${COMPONENT_LIB} PRIVATE ${arithmetic} VAR_ARRAYS EXPORT=)
target_compile_options(${COMPONENT_LIB} PRIVATE -fwrapv)
set_source_files_properties("${dsp}/fftwrap.c" PROPERTIES COMPILE_OPTIONS "-Wno-error=maybe-uninitialized")
if(NYX_SPEEX_FIXED_CUE)
    target_sources(${COMPONENT_LIB} PRIVATE "cue.cpp")
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../portable")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NYX_SPEEX_FIXED_CUE)
endif()
