set(src "${CMAKE_CURRENT_LIST_DIR}/../../../../.local/wakeword/c161-first-command-v5/lexical-echo/speexdsp-1.2.1")
set(dsp "${src}/libspeexdsp")
if(NYX_SPEEX_FLOAT)
    set(fft_src "${dsp}/smallft.c")
    set(arithmetic FLOATING_POINT USE_SMALLFT)
else()
    set(fft_src "${dsp}/kiss_fft.c" "${dsp}/kiss_fftr.c")
    set(arithmetic FIXED_POINT USE_KISS_FFT)
endif()
set(mdf "${dsp}/mdf.c")
if(NYX_SPEEX_COLD_GUARD AND NOT (NYX_SPEEX_HOLD AND NYX_SPEEX_FIXED_CUE))
    message(FATAL_ERROR "The cold-start reference guard requires the foreground hold and v5 matcher")
endif()
if(NYX_SPEEX_HOLD)
    file(READ "${mdf}" original)
    string(REPLACE "if (update_foreground)" "if (update_foreground && !nyx_local_demo_active())" held "${original}")
    if(held STREQUAL original)
        message(FATAL_ERROR "Could not locate the SpeexDSP foreground transfer")
    endif()
    set(mdf "${CMAKE_CURRENT_BINARY_DIR}/mdf_hold.c")
    file(WRITE "${mdf}" "extern int nyx_local_demo_active(void);\n${held}")
    if(NYX_SPEEX_COLD_GUARD)
        file(APPEND "${mdf}" "\nint nyx_echo_adapted(const SpeexEchoState *st) { return st->adapted; }\n")
    endif()
endif()
idf_component_register(SRCS "main.c" "${mdf}" "${dsp}/fftwrap.c" ${fft_src}
                       PRIV_INCLUDE_DIRS "." "${src}/include" "${dsp}" PRIV_REQUIRES esp_timer spi_flash heap)
target_compile_definitions(${COMPONENT_LIB} PRIVATE ${arithmetic} VAR_ARRAYS EXPORT=)
target_compile_options(${COMPONENT_LIB} PRIVATE -fwrapv)
set_source_files_properties("${dsp}/fftwrap.c" PROPERTIES COMPILE_OPTIONS "-Wno-error=maybe-uninitialized")
if(NYX_SPEEX_FIXED_CUE)
    target_sources(${COMPONENT_LIB} PRIVATE "cue.cpp")
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../portable")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NYX_SPEEX_FIXED_CUE)
endif()
if(NYX_SPEEX_HOLD)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NYX_SPEEX_HOLD)
endif()
if(NYX_SPEEX_COLD_GUARD)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NYX_SPEEX_COLD_GUARD)
endif()
if(NYX_SPEEX_PREPROCESS)
    target_sources(${COMPONENT_LIB} PRIVATE "${dsp}/preprocess.c" "${dsp}/filterbank.c")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NYX_SPEEX_PREPROCESS)
endif()
